const SHEET_NAME="Count"; const LOG_SHEET_NAME="CountHistory"; const TOKEN_SHEET_NAME="AdminSessions"; const DEFAULT_ADMIN_PASSWORD="123456"; function setup(){ const ss=SpreadsheetApp.getActiveSpreadsheet(); let sh=ss.getSheetByName(SHEET_NAME)||ss.insertSheet(SHEET_NAME); sh.getRange("A1:E1").setValues([["Print Count","PDF Count","Total Count","Last Action","Last Time"]]); if(sh.getRange("A2").getValue()==="") sh.getRange("A2").setValue(0); if(sh.getRange("B2").getValue()==="") sh.getRange("B2").setValue(0); sh.getRange("C2").setValue(Number(sh.getRange("A2").getValue())+Number(sh.getRange("B2").getValue())); let log=ss.getSheetByName(LOG_SHEET_NAME)||ss.insertSheet(LOG_SHEET_NAME); if(log.getLastRow()===0) log.getRange("A1:D1").setValues([["Date","Type","Time","Action"]]); let tok=ss.getSheetByName(TOKEN_SHEET_NAME)||ss.insertSheet(TOKEN_SHEET_NAME); if(tok.getLastRow()===0) tok.getRange("A1:C1").setValues([["Token","Created","Expires"]]); const p=PropertiesService.getScriptProperties(); if(!p.getProperty("ADMIN_PASSWORD")) p.setProperty("ADMIN_PASSWORD",DEFAULT_ADMIN_PASSWORD); } function doGet(e){ try{ setup(); const action=String((e&&e.parameter&&e.parameter.action)||"").toLowerCase(); if(action==="history") return json_(getHistory_(String(e.parameter.date||""))); return json_(getToday_()); }catch(err){ return json_({ok:false,error:String(err)}); } } function doPost(e){ try{ setup(); const data=JSON.parse((e.postData&&e.postData.contents)||"{}"); const action=String(data.action||"").toLowerCase(); if(action==="increment") return json_(increment_(String(data.type||"").toLowerCase())); if(action==="login") return json_(login_(String(data.password||""))); if(action==="deleteall"||action==="delete_all") return json_(deleteAll_(String(data.token||""))); return json_({ok:false,error:"Invalid action: "+action}); }catch(err){ return json_({ok:false,error:String(err)}); } } function increment_(type){ if(!["print","pdf"].includes(type)) return {ok:false,error:"Invalid type"}; const lock=LockService.getScriptLock(); lock.waitLock(10000); try{ const ss=SpreadsheetApp.getActiveSpreadsheet(); const sh=ss.getSheetByName(SHEET_NAME); const log=ss.getSheetByName(LOG_SHEET_NAME); let print=Number(sh.getRange("A2").getValue())||0; let pdf=Number(sh.getRange("B2").getValue())||0; if(type==="print") print++; else pdf++; const total=print+pdf; const now=new Date(); sh.getRange("A2:E2").setValues([[print,pdf,total,type,now]]); log.appendRow([dateKey_(now),type,now,"increment"]); return {ok:true,print:print,pdf:pdf,total:total,date:dateKey_(now)}; }finally{ lock.releaseLock(); } } function getToday_(){ const sh=SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); const print=Number(sh.getRange("A2").getValue())||0; const pdf=Number(sh.getRange("B2").getValue())||0; return {ok:true,print:print,pdf:pdf,total:print+pdf,date:dateKey_(new Date())}; } function getHistory_(dateText){ if(!/^\\d{4}-\\d{2}-\\d{2}$/.test(dateText)) return {ok:false,error:"Invalid date"}; const log=SpreadsheetApp.getActiveSpreadsheet().getSheetByName(LOG_SHEET_NAME); let print=0,pdf=0; if(log&&log.getLastRow()>=2){ const values=log.getRange(2,1,log.getLastRow()-1,4).getValues(); values.forEach(function(r){ if(String(r[0])===dateText){ if(String(r[1]).toLowerCase()==="print") print++; if(String(r[1]).toLowerCase()==="pdf") pdf++; } }); } return {ok:true,date:dateText,print:print,pdf:pdf,total:print+pdf}; } function login_(password){ const correct=PropertiesService.getScriptProperties().getProperty("ADMIN_PASSWORD")||DEFAULT_ADMIN_PASSWORD; if(!password||password!==correct) return {ok:false,error:"Wrong password"}; const token=Utilities.getUuid(); const now=new Date(); const expires=new Date(Date.now()+30*60*1000); SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TOKEN_SHEET_NAME).appendRow([token,now,expires]); return {ok:true,token:token}; } function deleteAll_(token){ if(!validToken_(token)) return {ok:false,error:"Unauthorized or expired Admin session"}; const lock=LockService.getScriptLock(); lock.waitLock(10000); try{ const ss=SpreadsheetApp.getActiveSpreadsheet(); const log=ss.getSheetByName(LOG_SHEET_NAME); if(log&&log.getLastRow()>1){ log.getRange(2,1,log.getLastRow()-1,Math.max(4,log.getLastColumn())).clearContent(); } const sh=ss.getSheetByName(SHEET_NAME); if(sh) sh.getRange("A2:E2").setValues([[0,0,0,"delete all",new Date()]]); const tok=ss.getSheetByName(TOKEN_SHEET_NAME); if(tok&&tok.getLastRow()>1) tok.getRange(2,1,tok.getLastRow()-1,3).clearContent(); SpreadsheetApp.flush(); return {ok:true,print:0,pdf:0,total:0,message:"All counts and history deleted"}; }finally{ lock.releaseLock(); } } function validToken_(token){ if(!token) return false; const tok=SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TOKEN_SHEET_NAME); if(!tok||tok.getLastRow()<2) return false; const values=tok.getRange(2,1,tok.getLastRow()-1,3).getValues(); const now=new Date(); return values.some(function(r){ return String(r[0])===token && r[2] instanceof Date && r[2]>now; }); } function dateKey_(d){ return Utilities.formatDate(d,Session.getScriptTimeZone(),"yyyy-MM-dd"); } function json_(obj){ return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
